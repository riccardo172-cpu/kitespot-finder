async function getForecast(spot) { const date = getSelectedDate(); const day = dateISO(date); // Chiediamo a Open-Meteo i prossimi 16 giorni. // Poi selezioniamo solamente il giorno scelto. const url = "https://api.open-meteo.com/v1/forecast" + "?latitude=" + encodeURIComponent(spot.lat) + "&longitude=" + encodeURIComponent(spot.lon) + "&hourly=" + "wind_speed_10m," + "wind_direction_10m," + "wind_gusts_10m" + "&wind_speed_unit=kn" + "&timezone=auto" + "&forecast_days=16"; const response = await fetch(url); if (!response.ok) { console.error( "Open-Meteo HTTP error:", response.status, spot.name ); return null; } const data = await response.json(); if ( !data.hourly || !Array.isArray(data.hourly.time) ) { console.error( "Risposta Open-Meteo non valida:", spot.name, data ); return null; } const hours = []; for ( let i = 0; i < data.hourly.time.length; i++ ) { const time = data.hourly.time[i]; // Prendiamo solamente il giorno selezionato. if (!time.startsWith(day)) { continue; } const hour = Number( time.substring(11, 13) ); // Consideriamo dalle 08:00 alle 20:00 comprese. if (hour < 8 || hour > 20) { continue; } const wind = Number( data.hourly.wind_speed_10m[i] ); const gust = Number( data.hourly.wind_gusts_10m[i] ); const direction = Number( data.hourly.wind_direction_10m[i] ); if (!Number.isFinite(wind)) { continue; } hours.push({ hour: hour, wind: wind, gust: Number.isFinite(gust) ? gust : 0, direction: Number.isFinite(direction) ? direction : 0 }); } if (hours.length === 0) { console.warn( "Nessun dato orario per:", spot.name, day ); return null; } // Ore che raggiungono la soglia selezionata. const qualifying = hours.filter( h => h.wind >= selectedWind ); // Vento massimo della giornata. let maxWindHour = hours[0]; let maxGust = 0; for (const hour of hours) { if (hour.wind > maxWindHour.wind) { maxWindHour = hour; } if (hour.gust > maxGust) { maxGust = hour.gust; } } // Se nessuna ora raggiunge la soglia, // lo spot viene escluso. if (qualifying.length === 0) { return null; } // Costruiamo gli intervalli consecutivi. const intervals = []; let start = qualifying[0]; let previous = qualifying[0]; for ( let i = 1; i < qualifying.length; i++ ) { const current = qualifying[i]; if ( current.hour - previous.hour > 1 ) { intervals.push({ start: start.hour, end: previous.hour }); start = current; } previous = current; } intervals.push({ start: start.hour, end: previous.hour }); return { date: date, dateLabel: formatDate(date), maxWind: maxWindHour.wind, maxGust: maxGust, direction: maxWindHour.direction, intervals: intervals, qualifying: qualifying }; }
