function getCurrentPosition(){ const status=document.getElementById("status"); status.className="status"; status.textContent="Rilevamento posizione..."; if(!navigator.geolocation){ status.className="status error"; status.textContent="Geolocalizzazione non disponibile."; return; } navigator.geolocation.getCurrentPosition( function(position){ searchPosition={ lat:position.coords.latitude, lon:position.coords.longitude, name:"Posizione attuale" }; updateLocation(); status.className="status success"; status.textContent="Posizione acquisita."; }, function(){ status.className="status error"; status.textContent= "Impossibile ottenere la posizione."; }, { enableHighAccuracy:true, timeout:10000, maximumAge:300000 } ); }
